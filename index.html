function fetchData() { // เปลี่ยน URL มาดึงที่ Root เพื่อให้เจอค่า current_watt และ total_baht fetch("https://smart-money-meter-default-rtdb.asia-southeast1.firebasedatabase.app/.json") .then(res => res.json()) .then(data => { if(!data) return; // อัปเดตตัวเลข (ตอนนี้จะเจอค่า 200 และ 75 ตามใน Firebase แล้ว) document.getElementById('dash-watt').innerText = data.current_watt || 0; document.getElementById('dash-baht').innerText = (data.total_baht || 0).toFixed(2); const ai = document.getElementById('ai-status'); // เช็คเงื่อนไขงบประมาณจาก Path: test_device -> budget_limit if (data.test_device && data.total_baht >= data.test_device.budget_limit) { ai.innerText = "⚠️ เกินงบประมาณ!"; ai.style.background = "var(--danger)"; } else { ai.style.background = "var(--info)"; detect(data.current_watt); } }); }
